cmake_minimum_required(VERSION 3.25)
project(FileManagerJpegExperiment LANGUAGES CXX)

# Research target only. No application/GUI.Forms dependency or install rule.
option(FILE_MANAGER_JPEG_CMAKE_PACKAGE "Use an independently built static codec package" OFF)
if(FILE_MANAGER_JPEG_CMAKE_PACKAGE)
    find_package(libjpeg-turbo 3.2.0 EXACT CONFIG REQUIRED)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(TURBOJPEG REQUIRED IMPORTED_TARGET libturbojpeg=3.2.0)
endif()
add_library(file_manager_jpeg_orientation STATIC exif_orientation.cpp)
target_compile_features(file_manager_jpeg_orientation PUBLIC cxx_std_20)
set_target_properties(file_manager_jpeg_orientation PROPERTIES CXX_EXTENSIONS OFF)
add_executable(file_manager_jpeg_orientation_tests exif_orientation_tests.cpp)
target_link_libraries(file_manager_jpeg_orientation_tests PRIVATE file_manager_jpeg_orientation)
set_target_properties(file_manager_jpeg_orientation_tests PROPERTIES CXX_EXTENSIONS OFF)
add_executable(file_manager_jpeg_experiment jpeg_experiment.cpp)
if(WIN32)
    target_sources(file_manager_jpeg_experiment PRIVATE wic_decode.cpp)
    target_compile_definitions(file_manager_jpeg_experiment PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(file_manager_jpeg_experiment PRIVATE ole32 windowscodecs shlwapi)
endif()
target_link_libraries(file_manager_jpeg_experiment PRIVATE file_manager_jpeg_orientation)
if(FILE_MANAGER_JPEG_CMAKE_PACKAGE)
    target_link_libraries(file_manager_jpeg_experiment PRIVATE libjpeg-turbo::turbojpeg-static)
else()
    target_link_libraries(file_manager_jpeg_experiment PRIVATE PkgConfig::TURBOJPEG)
endif()
target_compile_features(file_manager_jpeg_experiment PRIVATE cxx_std_20)
set_target_properties(file_manager_jpeg_experiment PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(file_manager_jpeg_experiment PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(file_manager_jpeg_experiment PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
if(MSVC)
    target_compile_options(file_manager_jpeg_orientation PRIVATE /W4 /WX /permissive-)
    target_compile_options(file_manager_jpeg_orientation_tests PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(file_manager_jpeg_orientation PRIVATE -Wall -Wextra -Werror -Wpedantic)
    target_compile_options(file_manager_jpeg_orientation_tests PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
enable_testing()
add_test(NAME file_manager_jpeg_orientation_tests COMMAND file_manager_jpeg_orientation_tests)
add_test(NAME file_manager_jpeg_embedded_orientation_tests COMMAND file_manager_jpeg_experiment --verify-only)
if(WIN32)
    add_test(NAME file_manager_jpeg_wic_control_tests COMMAND file_manager_jpeg_experiment --verify-wic)
endif()
