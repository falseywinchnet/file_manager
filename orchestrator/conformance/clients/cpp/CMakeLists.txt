cmake_minimum_required(VERSION 3.20)
project(fileman_orchestrator_cpp_client LANGUAGES CXX)

add_library(fileman_orchestrator_client STATIC src/client.cpp)
add_library(fileman::orchestrator_client ALIAS fileman_orchestrator_client)
target_include_directories(fileman_orchestrator_client PUBLIC include)
target_compile_features(fileman_orchestrator_client PUBLIC cxx_std_17)
if(WIN32)
    target_link_libraries(fileman_orchestrator_client PRIVATE advapi32)
    target_compile_definitions(fileman_orchestrator_client PRIVATE _WIN32_WINNT=0x0602)
endif()
set_target_properties(fileman_orchestrator_client PROPERTIES CXX_EXTENSIONS OFF)

if(MSVC)
    target_compile_options(fileman_orchestrator_client PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(fileman_orchestrator_client PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()

option(FILEMAN_ORCHESTRATOR_BUILD_PLATFORM_CHECKS "Build snapshot and platform availability checks" OFF)
if(FILEMAN_ORCHESTRATOR_BUILD_PLATFORM_CHECKS)
    enable_testing()
    add_executable(orchestrator-cpp-platform-checks src/platform_checks.cpp)
    target_link_libraries(orchestrator-cpp-platform-checks PRIVATE fileman::orchestrator_client)
    add_test(NAME orchestrator-cpp-platform-checks COMMAND orchestrator-cpp-platform-checks)
endif()

option(FILEMAN_ORCHESTRATOR_BUILD_PROBE "Build the standalone conformance probe" ON)
option(FILEMAN_ORCHESTRATOR_BUILD_SEARCH_CHECKS "Build search response projection checks" OFF)
if(FILEMAN_ORCHESTRATOR_BUILD_SEARCH_CHECKS)
    enable_testing()
    add_executable(orchestrator-cpp-search-checks src/search_projection_checks.cpp)
    target_link_libraries(orchestrator-cpp-search-checks PRIVATE fileman::orchestrator_client)
    if(MSVC)
        target_compile_options(orchestrator-cpp-search-checks PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(orchestrator-cpp-search-checks PRIVATE -Wall -Wextra -Werror -Wpedantic)
    endif()
    add_test(NAME orchestrator-cpp-search-checks COMMAND orchestrator-cpp-search-checks)
endif()

if(FILEMAN_ORCHESTRATOR_BUILD_PROBE)
    add_executable(orchestrator-cpp-client src/main.cpp)
    target_link_libraries(orchestrator-cpp-client PRIVATE fileman::orchestrator_client)
endif()
