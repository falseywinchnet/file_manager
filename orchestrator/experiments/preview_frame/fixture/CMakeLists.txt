cmake_minimum_required(VERSION 3.25)
project(PreviewFrameFixture LANGUAGES CXX)
add_executable(preview_frame_fixture producer.cpp)
target_compile_features(preview_frame_fixture PRIVATE cxx_std_20)
set_target_properties(preview_frame_fixture PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(preview_frame_fixture PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(preview_frame_fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
