cmake_minimum_required(VERSION 3.25)
project(PreviewProcessLaboratory LANGUAGES CXX)

# Native API experiment only; no runtime, codec, GUI or install dependency.
add_executable(preview_process main.cpp)
target_compile_features(preview_process PRIVATE cxx_std_20)
set_target_properties(preview_process PROPERTIES CXX_EXTENSIONS OFF)
if(WIN32)
    target_sources(preview_process PRIVATE windows_process.cpp)
    target_compile_definitions(preview_process PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
else()
    target_sources(preview_process PRIVATE posix_process.cpp)
endif()
if(MSVC)
    target_compile_options(preview_process PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(preview_process PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME preview_process_lifecycle COMMAND preview_process --lifecycle)
add_test(NAME preview_process_memory COMMAND preview_process --memory-probe)
set_tests_properties(preview_process_lifecycle preview_process_memory PROPERTIES TIMEOUT 30)
# An explicitly unavailable Mac address-space limit is evidence, not a pass.
set_tests_properties(preview_process_memory PROPERTIES SKIP_RETURN_CODE 77)
if(APPLE)
    add_test(NAME preview_process_mac_headroom COMMAND preview_process --headroom-probe)
    set_tests_properties(preview_process_mac_headroom PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
endif()
